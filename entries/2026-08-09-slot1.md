# Streaming LLM Responses with Cancellation Support

When building AI chat interfaces, streaming tokens from the model improves perceived latency. The client should be able to cancel in-flight requests without leaking connections or leaving partial state.

```tsx
// Client-side hook for streaming with abort support
function useStreamingChat() {
  const [messages, setMessages] = useState<Message[]>([]);
  const abortRef = useRef<AbortController | null>(null);

  const sendMessage = async (content: string) => {
    abortRef.current?.abort();
    const controller = new AbortController();
    abortRef.current = controller;

    const userMsg: Message = { role: "user", content };
    const assistantMsg: Message = { role: "assistant", content: "" };
    setMessages((m) => [...m, userMsg, assistantMsg]);

    try {
      const res = await fetch("/api/chat", {
        method: "POST",
        body: JSON.stringify({ messages: [...messages, userMsg] }),
        signal: controller.signal,
      });

      if (!res.ok) throw new Error(await res.text());

      const reader = res.body!.getReader();
      const decoder = new TextDecoder();
      let buffer = "";

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        buffer += decoder.decode(value, { stream: true });
        const lines = buffer.split("\n");
        buffer = lines.pop()!;

        for (const line of lines) {
          if (line.startsWith("data: ")) {
            const token = line.slice(6);
            setMessages((m) => {
              const copy = [...m];
              copy[copy.length - 1] = {
                ...copy[copy.length - 1],
                content: copy[copy.length - 1].content + token,
              };
              return copy;
            });
          }
        }
      }
    } catch (e) {
      if (e instanceof DOMException && e.name === "AbortError") return;
      setMessages((m) => m.slice(0, -1));
      throw e;
    }
  };

  const cancel = () => abortRef.current?.abort();

  return { messages, sendMessage, cancel };
}
```

- Use `AbortController` to cancel in-flight streams cleanly; the server should respect the signal and stop generation.
- Buffer and parse SSE frames incrementally to handle partial chunks; never assume a single `read()` returns a complete frame.
- On abort, remove the optimistic assistant message from state rather than leaving a truncated response.
