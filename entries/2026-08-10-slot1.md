Set per-call LLM timeouts that respect your total request budget and fail fast on slow providers. A single slow call should not block the whole request.

```python
import asyncio
from dataclasses import dataclass
from typing import Optional

@dataclass
class TimeoutBudget:
    total_ms: int
    per_call_ms: int
    per_tool_ms: int

    def child(self, name: str) -> "TimeoutBudget":
        return TimeoutBudget(
            total_ms=self.total_ms,
            per_call_ms=min(self.per_call_ms, self.per_tool_ms),
            per_tool_ms=self.per_tool_ms,
        )

async def call_llm_with_budget(prompt: str, budget: TimeoutBudget) -> str:
    try:
        return await asyncio.wait_for(
            _call_llm(prompt),
            timeout=budget.per_call_ms / 1000,
        )
    except asyncio.TimeoutError:
        raise LLMTimeoutError(f"LLM call exceeded {budget.per_call_ms}ms budget")

async def run_agent(user_input: str, budget: TimeoutBudget) -> str:
    start = asyncio.get_event_loop().time()
    try:
        return await asyncio.wait_for(
            _agent_loop(user_input, budget.child("agent")),
            timeout=budget.total_ms / 1000,
        )
    except asyncio.TimeoutError:
        elapsed = int((asyncio.get_event_loop().time() - start) * 1000)
        raise AgentTimeoutError(f"Agent exceeded total budget of {budget.total_ms}ms after {elapsed}ms")
```

- Cascade budgets: total request budget, per-LLM-call budget, per-tool budget. Each layer enforces its own ceiling.
- Fail fast on slow providers instead of letting one hung call consume the entire request window.
- Log the elapsed time and which budget tier triggered so you can tune limits from production data.
